CREATE TABLE cab_owners (
    id INT AUTO_INCREMENT PRIMARY KEY,
    name VARCHAR(100),
    username VARCHAR(50) UNIQUE,
    mobile VARCHAR(15),
    email VARCHAR(100),
    password VARCHAR(255),
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE notifications (
    id INT AUTO_INCREMENT PRIMARY KEY,
    owner_id INT,
    message TEXT NOT NULL,
    is_read TINYINT(1) DEFAULT 0,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE cab_cars (
  id INT AUTO_INCREMENT PRIMARY KEY,
  owner_id INT,
  title VARCHAR(100),
  car_name VARCHAR(100),
  model VARCHAR(100),
  reg_number VARCHAR(100) UNIQUE,
  seating_capacity INT,
  rent_per_day DECIMAL(10,2),
  rate_per_km DECIMAL(10,2),
  security_deposit DECIMAL(10,2),
  location VARCHAR(100),
  is_available TINYINT(1) DEFAULT 1,
  car_details TEXT,
  car_image VARCHAR(255),
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE cab_users (
  id INT AUTO_INCREMENT PRIMARY KEY,
  name VARCHAR(100),
  username VARCHAR(50) UNIQUE,
  email VARCHAR(100),
  mobile VARCHAR(20),
  password VARCHAR(255),
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE cab_bookings (
    id INT AUTO_INCREMENT PRIMARY KEY,
    user_id INT,
    offline_customer_id INT,
    car_id INT NOT NULL,
    start_date DATE NOT NULL,
    end_date DATE NOT NULL,
    pickup_location VARCHAR(255),
    drop_location VARCHAR(255),
    pickup_time TIME,
    drop_time TIME,
    mobile VARCHAR(15),
    type VARCHAR(50) NOT NULL,
    estimation DECIMAL(10,2) NOT NULL,
    final_cost DECIMAL(10,2),
    remarks TEXT,
    status VARCHAR(50) DEFAULT 'Pending',
    payment_status VARCHAR(50) DEFAULT 'Unpaid', 
    payment_mode VARCHAR(50),                   
    payment_timestamp TIMESTAMP NULL,           
    transaction_id VARCHAR(100),                 
    payment_remarks TEXT,                       
    
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);


CREATE TABLE user_notifications (
    id INT AUTO_INCREMENT PRIMARY KEY,
    user_id INT,
    message TEXT NOT NULL,
    is_read TINYINT(1) DEFAULT 0,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE offline_customers (
    id INT AUTO_INCREMENT PRIMARY KEY,
    name VARCHAR(100),
    mobile VARCHAR(20),
    email VARCHAR(100),
    address TEXT,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);
